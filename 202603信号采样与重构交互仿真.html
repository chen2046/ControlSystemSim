对应 MATLAB 程序

% 信号采样与重建仿真 % 参数设置 F = 5; % 信号频率 (Hz) Fs = 20; % 采样频率 (Hz) T = 1/F; % 信号周期 t = 0:0.001:2*T; % 细时间向量（用于绘制连续信号） ts = 0:1/Fs:2*T; % 采样时间点 % 原始信号（例如余弦信号） x = cos(2*pi*F*t); xs = cos(2*pi*F*ts); % 重建（理想低通滤波器） xr = zeros(size(t)); for n = 1:length(ts) xr = xr + xs(n) * sinc(Fs*(t - ts(n))); end % 绘图 figure; subplot(2,1,1); plot(t, x, 'b-', ts, xs, 'ro'); title('原始信号与采样点'); legend('连续信号','采样点'); xlabel('时间 (s)'); subplot(2,1,2); plot(t, x, 'b-', t, xr, 'g--'); title('重建信号'); legend('原始','重建'); xlabel('时间 (s)');
