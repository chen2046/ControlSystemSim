卷积结果 y(t)

MATLAB 程序

% 信号卷积 t = -5:0.001:5; T1 = 2; A1 = 1; % 信号1参数 T2 = 2; A2 = 1; % 信号2参数 % 矩形脉冲信号 x = A1 * double(abs(t) <= T1/2); h = A2 * double(abs(t) <= T2/2); % 数值卷积 dt = 0.001; y = conv(x, h) * dt; t_conv = (0:length(y)-1)*dt - (length(x)-1)/2*dt; figure; subplot(3,1,1); plot(t, x, 'b-', 'LineWidth', 2); title('信号 x(t)'); ylabel('幅度'); grid on; subplot(3,1,2); plot(t, h, 'r-', 'LineWidth', 2); title('信号 h(t)'); ylabel('幅度'); grid on; subplot(3,1,3); plot(t_conv, y, 'g-', 'LineWidth', 2); title('卷积结果 y(t) = x(t) * h(t)'); xlabel('时间 t'); ylabel('幅度'); grid on;
