% 信号傅里叶级数合成 f0 = 5; % 基频 (Hz) N = 10; % 谐波次数 A = 1; % 幅度 duty = 0.5; % 占空比 (0-1) T = 1/f0; t = 0:0.0001:3*T; % 初始化信号 x = zeros(size(t)); % 傅里叶系数及合成 for k = 1:N if strcmp('square', 'square') bk = (4*A/(k*pi)) * sin(k*pi*duty); % 方波系数 (仅奇次) else % triangle if mod(k,2) == 1 % 仅奇次谐波 bk = (8*A/(k^2*pi^2)) * sin(k*pi*duty); else bk = 0; end end x = x + bk * sin(2*pi*k*f0*t); end % 绘图 figure; subplot(3,1,1); plot(t, x, 'b-', 'LineWidth', 2); title(['合成波形: ', '方波', ' (N=', num2str(N), ')']); xlabel('时间 t'); ylabel('幅度'); grid on; % 幅频特性 k = 1:N; figure; stem(k, abs([1, 0, 0.424413, 0, 0.090946, 0, 0.040484, 0, 0.02242, 0]), 'filled'); title('幅频特性'); xlabel('谐波次数 k'); ylabel('|b_k|'); grid on;

% LTI系统复频域分析 K = 1.0; % 增益 poles = [-1.00]; % 极点 zeros = []; % 零点 % 构造传递函数 num = poly(zeros); % 零点多项式 den = poly(poles); % 极点多项式 sys = tf(K * num, den); % 零极点图 figure; pzmap(sys); title('零极点分布图'); grid on; xlabel('实部'); ylabel('虚部'); % Bode图 figure; bode(sys); grid on; % 稳定性判断 if all(real(poles) < 0) disp('系统稳定'); else disp('系统不稳定'); end
